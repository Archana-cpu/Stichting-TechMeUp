```typescript
// Organization routes: Check membership + role permissions
if (pathname.startsWith('/org/')) {
  const membership = await checkOrgMembership(userId, orgSlug)
  if (!membership) return redirect('/unauthorized')

  // Route-specific permission checks
  if (pathname.includes('/members') && !hasPermission(membership.role, 'members:manage')) {
    return redirect('/unauthorized')
  }
}

// Admin routes: Require PLATFORM_ADMIN role
if (pathname.startsWith('/admin/')) {
  if (session.role !== 'PLATFORM_ADMIN') {
    return redirect('/unauthorized')
  }
}
```

**Mobile App Navigation Guards** (`apps/mobile/app/_layout.tsx`):
```typescript
const inOrgGroup = segments[0] === 'org'

if (inOrgGroup) {
  const orgSlug = segments[1]
  const hasAccess = await checkOrgAccess(session.userId, orgSlug)
  if (!hasAccess) router.replace('/unauthorized')
}
```

**API Permission Middleware** (`apps/api/src/middleware/auth.ts`):
```typescript
export function requireOrgPermission(permission: Permission) {
  return async (c: Context, next: Next) => {
    const membership = await getOrgMembership(userId, slug)
    if (!hasPermission(membership.role, permission)) {
      return c.json({ error: 'Insufficient permissions' }, 403)
    }
    await next()
  }
}
```
